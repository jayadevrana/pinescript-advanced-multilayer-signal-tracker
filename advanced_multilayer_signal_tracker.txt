f_clamp(value, min_val, max_val) => math.max(min_val, math.min(value, max_val))
f_bbp(_s, _len, _mult) =>
    [_, upper, lower] = ta.bb(_s, _len, _mult)
    bb_range = math.max(upper - lower, 1e-10)
    (_s - lower) / bb_range
f_get_bb_region(bbp_val) =>
	if bbp_val > 1
		"A"
	else if bbp_val == 1
		"AA"
	else if bbp_val < 0
		"D"
	else if bbp_val == 0
		"DD"
	else

f_get_cci_region(cci_val) =>
	if cci_val > 100
		"A"
	else if cci_val == 100
		"AA"
	else if cci_val >= 0 and cci_val < 100
		"B"
	else if cci_val < 0 and cci_val > -100
		"C"
	else if cci_val < -100
		"D"
	else
		"DD"

f_round_max(value, round_value, delta) =>
	decimal_part = value % 1
	result = decimal_part >= round_value ? math.ceil(value) + 1 + delta : math.ceil(value) + delta
	result
f_round_min(value, round_value, delta) =>
	decimal_part = value % 1
	result = decimal_part < 1 - round_value ? math.floor(value) - 1 - delta : math.floor(value) - delta
	result

f_python_indicators(src, high, low, close, round_val_input, delta_input) =>
	var float original_high = 0, var float original_low = 9999999
	float MAXA = 0.0, float MINA = 0.0, var bool is_shadow_bar = false, int breakout_signal = 0
	if high <= original_high and low >= original_low
		is_shadow_bar := true, RA = (0.309 * (high - low)) + close, SA = math.abs((0.309 * (high - low)) - close)
		MAXA := math.max(math.max(math.max(high, RA), original_high), MAXA[1]), MINA := math.min(math.min(math.min(low, SA), original_low), MINA[1])
	else
		if is_shadow_bar[1]
			if high > original_high
				breakout_signal := 1
			else if low < original_low
				breakout_signal := -1
		is_shadow_bar := false, original_high := high, original_low := low
		RA = (0.309 * (high - low)) + close, SA = math.abs((0.309 * (high - low)) - close)
		MAXA := math.max(high, RA), MINA := math.min(low, SA)
	float MAXAr = f_round_max(MAXA, round_val_input, delta_input), float MINAr = f_round_min(MINA, round_val_input, delta_input)
	[MAXAr, MINAr, is_shadow_bar, original_high, original_low, breakout_signal, MAXA, MINA]

enum BreakoutState
	NONE
	JUST_UP
	JUST_DOWN
	UP
	DOWN
	DU_1
	DU_2
	UD_1
	UD_2


f_cci_breakout_logic(cci_val, constraint, maxar, minar, high_, low_) =>
	var float ref = maxar, var int last_dir = 0
	int breakout_event = 0, var int breakout_bar = 0
	var state = BreakoutState.NONE
	if ta.crossover(cci_val, constraint)
		ref := maxar, state := BreakoutState.JUST_UP
	if ta.crossunder(cci_val, constraint)
		ref := minar, state := BreakoutState.JUST_DOWN

	if state == BreakoutState.JUST_UP
		state := BreakoutState.UP
	else if state == BreakoutState.JUST_DOWN
		state := BreakoutState.DOWN

	if state == BreakoutState.UP
		if high_ > ref
			breakout_event := 1, last_dir := 1, breakout_bar := bar_index, state := BreakoutState.NONE
	else if state == BreakoutState.DOWN
		if low_ < ref
			breakout_event := -1, last_dir := -1, breakout_bar := bar_index, state := BreakoutState.NONE

	if state == BreakoutState.UP
		if (cci_crossunder_B or cci_crossunder_A) and constraint == -100
			state := BreakoutState.UD_1
	else if state == BreakoutState.DOWN
		if (cci_crossover_B or cci_crossover_C) and constraint == 100
			state := BreakoutState.DU_1
	else if state == BreakoutState.UD_1
		if cci_crossover_C or cci_crossover_B
			ref := maxar, state := BreakoutState.UP
	else if state == BreakoutState.DU_1
		if cci_crossunder_C or cci_crossunder_B
			ref := minar, state := BreakoutState.DOWN

	[breakout_event, ref, last_dir, breakout_bar]

f_bbp_breakout_logic(bbp_val, constraint, maxar, minar, high_, low_) =>
	var float ref = maxar, var int last_dir = 0
	int breakout_event = 0, var int breakout_bar = 0
	var state = BreakoutState.NONE
	if ta.crossover(bbp_val, constraint)
		ref := maxar, state := BreakoutState.JUST_UP
	if ta.crossunder(bbp_val, constraint)
		ref := minar, state := BreakoutState.JUST_DOWN

	if state == BreakoutState.JUST_UP
		state := BreakoutState.UP
	else if state == BreakoutState.JUST_DOWN
		state := BreakoutState.DOWN

	if state == BreakoutState.UP
		if high_ > ref
			breakout_event := 1, last_dir := 1, breakout_bar := bar_index, state := BreakoutState.NONE
	else if state == BreakoutState.DOWN
		if low_ < ref
			breakout_event := -1, last_dir := -1, breakout_bar := bar_index, state := BreakoutState.NONE

	if state == BreakoutState.UP
		if (bbp_crossunder_B or bbp_crossunder_A) and constraint == 0.0
			state := BreakoutState.UD_1
	else if state == BreakoutState.DOWN
		if (bbp_crossover_B or bbp_crossover_C) and constraint == 1.0
			state := BreakoutState.DU_1
	else if state == BreakoutState.UD_1
		if bbp_crossover_C or bbp_crossover_B
			ref := maxar, state := BreakoutState.UP
	else if state == BreakoutState.DU_1
		if (bbp_crossunder_C or bbp_crossunder_B)
			ref := minar, state := BreakoutState.DOWN

	[breakout_event, ref, last_dir, breakout_bar]

f_map_trade_type(trade_type) =>
    if str.length(trade_type) == 0
        ""
    else
        num_str = str.substring(trade_type, 0, str.length(trade_type) - 1)
        type_char = str.substring(trade_type, str.length(trade_type) - 1, str.length(trade_type))
        trade_num = str.tonumber(num_str)

        base_action = type_char == 'B' ? "Buy" : "Sell"
        modifier = ""

        if trade_num > 1
            if trade_num % 2 == 0 // Even: Exit
                num_reverses = (trade_num - 2) / 2
                // Replace loop with str.repeat
                modifier += str.repeat("Reverse ", int(num_reverses))
                modifier += "Exit"
            else // Odd: Reverse
                num_reverses = (trade_num - 1) / 2
                // Replace loop with str.repeat
                modifier += str.repeat("Reverse ", int(num_reverses))

        str.trim(base_action + " " + modifier)

var last_trade_type_arr = array.new_string(1, "")  // index 0 = last trade type
f_is_valid_trade_type(trade_type) =>
	if trade_type == ""
		false
	else
		string last_trade_type = array.get(last_trade_type_arr, 0)
		if last_trade_type == ""
			array.set(last_trade_type_arr, 0, trade_type)
		last_trade_type := array.get(last_trade_type_arr, 0)
		if trade_type != ""
			last_num_str = str.substring(last_trade_type, 0, str.length(last_trade_type) - 1)
			last_type_str = str.substring(last_trade_type, str.length(last_trade_type) - 1, str.length(last_trade_type))
			last_trade_num = str.tonumber(last_num_str)

			num_str = str.substring(trade_type, 0, str.length(trade_type) - 1)
			type_str = str.substring(trade_type, str.length(trade_type) - 1, str.length(trade_type))
			trade_num = str.tonumber(num_str)
			bool show = false
			if type_str == last_type_str or (last_trade_num == 1 and type_str == last_type_str) or last_trade_num % 2 == 0
				if trade_num == last_trade_num
					show := false
				else if trade_num == 1 and last_trade_num % 2 == 0
					array.set(last_trade_type_arr, 0, trade_type)
					show := true
				else if trade_num != last_trade_num and trade_num % 2 == 1 and trade_num == last_trade_num + 2
					array.set(last_trade_type_arr, 0, trade_type)
					show := true
				else if trade_num != last_trade_num and trade_num % 2 == 0 and trade_num == last_trade_num + 1
					array.set(last_trade_type_arr, 0, trade_type)
					show := true
				// log.info("trade_type: " + trade_type + " " + last_trade_type + " " + str.tostring(show) + str.tostring(trade_num) + " " + str.tostring(last_trade_num))
			else if type_str == "C" and last_trade_num == trade_num
				show := true
			show
		else
			false

// updated plotting helper that uses the global array for stacking (mutable inside function)
f_plotSignal(enable, signal_settings, signal_value, text_, tf, bullColor, bearColor, bullTextColor, bearTextColor, size, layer) =>
    // HTF Change Detection: Only proceed if it's the first bar of a new HTF period.
    is_new_htf_bar = timeframe.isintraday and (tf != timeframe.period) and (time(tf) != time(tf)[1])
    can_plot = timeframe.isdaily or (tf == timeframe.period) or is_new_htf_bar
	can_lookahead = tf == timeframe.period
	// Request the close of the signal's timeframe.
	// Use `barmerge.lookahead_on` to get the current (non-repainting) close if the signal TF is higher than the chart TF.
	float signal_tf_close = 0

	signal_tf_close := request.security(syminfo.tickerid, tf, close, lookahead = barmerge.lookahead_on)

	ma = f_get_ref_maxar()
	mi = f_get_ref_minar()

    if enable and can_plot
        // reset stacking when moving to a new bar
        if array.get(_sig_stack, 0) != bar_index
            array.set(_sig_stack, 0, bar_index)
            array.set(_sig_stack, 1, 0)
            array.set(_sig_stack, 2, 0)

		// vertical spacing step (ATR(14) with safe minimum)
		float step = math.max(ta.atr(4), syminfo.mintick * 50)

		if signal_value == 1 and signal_settings.show_bull
			int bcount = array.get(_sig_stack, 1)
			float y = high + (bcount * step) + ta.atr(20)
			string human_tf = f_human_tf(tf)
			string label_text = ""
			string trade_type_prefix = signal_settings.trade_type_bull == "" ? "" : signal_settings.trade_type_bull + ": "
			log.info("TradeType: " + signal_settings.trade_type_bull)

			if not f_is_valid_trade_type(signal_settings.trade_type_bull)
				trade_type_prefix := ""
			if text_ == "."
				label_text := text_ + "\n" + trade_type_prefix
			else
				label_text := text_ + " " + human_tf  + "\n" + trade_type_prefix
			string bull_alert_message = (signal_settings.trade_type_bull == "" ? "" : trade_type_prefix + ": ") + "\n" + text_
			string dynamic_part = "\n" + f_map_trade_type(signal_settings.trade_type_bull)
			if alert_include_ticker
				dynamic_part += "\n" + syminfo.tickerid
			if alert_include_timeframe
				dynamic_part += "\n on " + human_tf
			dynamic_part += "\n at " + str.format_time(time, "yyyy-MM-dd HH:mm", "GMT+05:30")
			if alert_include_open
				dynamic_part += "\nO:" + str.tostring(open)
			if alert_include_high
				dynamic_part += "\nH:" + str.tostring(high)
			if alert_include_low
				dynamic_part += "\nL:" + str.tostring(low)




			string trade_type = signal_settings.trade_type_bull
			if str.length(trade_type) > 0
				string num_str = str.substring(trade_type, 0, str.length(trade_type) - 1)
				float trade_num = str.tonumber(num_str)
				if not na(trade_num)
					if trade_num == 1
						dynamic_part += "\n" + str.tostring(ma)
					else if trade_num % 2 == 0 or alert_include_close
						dynamic_part += "\nC:" + str.tostring(signal_tf_close)

			if alert_include_maxar
				dynamic_part += "\n" + str.tostring(ma)
			if alert_include_minar
				dynamic_part += "\n" + str.tostring(mi)

			bull_alert_message += dynamic_part + (signal_settings.bull_msg == "" ? "" : "\n" + signal_settings.bull_msg)
			log.info(bull_alert_message)
			label.new(bar_index, y, label_text, xloc = xloc.bar_index, yloc = yloc.price, color = na, textcolor = bullColor, style = label.style_label_up, size = size)
			array.set(_sig_stack, 1, bcount + 1)
			alert(bull_alert_message, freq = alert.freq_once_per_bar)

		if signal_value == -1 and signal_settings.show_bear
			int scount = array.get(_sig_stack, 2)
			float y2 = low - (scount * step) - ta.atr(20)
			string human_tf = f_human_tf(tf)
			string label_text = ""
			string trade_type_prefix = signal_settings.trade_type_bear == "" ? "" : signal_settings.trade_type_bear + ": "
			log.info("TradeType: " + signal_settings.trade_type_bear)

			if not f_is_valid_trade_type(signal_settings.trade_type_bear)
				trade_type_prefix := ""
			if text_ == "."
				label_text := text_ + "\n" + trade_type_prefix
			else
				label_text := text_ + " " + human_tf + "\n" + trade_type_prefix
			string bear_alert_message = (signal_settings.trade_type_bear == "" ? "" : trade_type_prefix + ": ") + "\n" + text_
			string dynamic_part = "\n" + f_map_trade_type(signal_settings.trade_type_bear)
			if alert_include_ticker
				dynamic_part += "\n" + syminfo.tickerid
			if alert_include_timeframe
				dynamic_part += "\non " + human_tf
			dynamic_part += "\n at " + str.format_time(time, "yyyy-MM-dd HH:mm", "GMT+05:30")
			if alert_include_open
				dynamic_part += "\nO:" + str.tostring(open)
			if alert_include_high
				dynamic_part += "\nH:" + str.tostring(high)
			if alert_include_low
				dynamic_part += "\nL:" + str.tostring(low)

						dynamic_part += "\nC:" + str.tostring(signal_tf_close)

			if alert_include_maxar
				dynamic_part += "\n" + str.tostring(ma)
			if alert_include_minar
				dynamic_part += "\n" + str.tostring(mi)

			bear_alert_message += dynamic_part + (signal_settings.bear_msg == "" ? "" : " | " + signal_settings.bear_msg)
			log.info(bear_alert_message)
			label.new(bar_index, y2, label_text, xloc = xloc.bar_index, yloc = yloc.price, color = na, textcolor = bearColor, style = label.style_label_down, size = size)
